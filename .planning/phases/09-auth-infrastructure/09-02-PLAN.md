    3. Run the following SQL exactly as written:

    ```sql
    -- Create tool_permissions table
    create table if not exists public.tool_permissions (
      user_id   uuid not null references auth.users(id) on delete cascade,
      tool_slug text not null,
      created_at timestamptz default now(),
      primary key (user_id, tool_slug)
    );

    -- Enable RLS (MANDATORY — without this, any authenticated user can read all rows)
    alter table public.tool_permissions enable row level security;

    -- Policy: each user can only SELECT their own rows
    create policy "Users can view their own tool permissions"
      on public.tool_permissions for select
      to authenticated
      using ( (select auth.uid()) = user_id );
    ```
